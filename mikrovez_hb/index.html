Tantárgy Áttekintése

A tantárgy célja a mikrovezérlők alapjainak gyakorlati elsajátítása: bemeneti/kimeneti kezelése, egyszerű perifériavezérlés és kis rendszerek prototípus készítése Arduino platformon.

Projektek és Feladatok

Roulette gép (csapatprojekt)

A csapatunk (3 fő) az év végén egy működő roulette játékgépet épített: az eszközt egy Arduino Nano vezérli. A megjelenítés LED mátrixon keresztül történik, a hangot buzzer adja, a beavatkozás gombokkal működik. A LED-eket mátrix kapcsolással, tartós forrasztással rögzítettük, a házat 3D nyomtatással készítettük.

Főbb komponensek

  • Arduino Nano
  • LED mátrix (multiplexelt vezérlés)
  • Buzzer, nyomógombok
  • 3D nyomtatott doboz

Arduino Kód

Megtekintés: Arduino code.txt 


#define PushButton 12

#define Click A5
#define ZeroLed 11
int quarter = A1; 
int led; 
bool CCW = true; 
void setup() {
    for (int i = 2; i <= 11; i++) pinMode(i, OUTPUT);
    pinMode(PushButton, INPUT_PULLUP);
    for (int a = A1; a <= A5; a++) pinMode(a, OUTPUT); 
    for (int nowmber = 1; nowmber < 4; nowmber++)
        for (int i = 2; i <= 10; i++) { 
            digitalWrite(i, HIGH);
            delay(200 - nowmber * 40 - (i - 2) * 5);
            digitalWrite(i, LOW);
        }
    randomSeed(analogRead(A0)); 
    led = random(2, 11); quarter = random(A1, A5); 
    for (int a = A1; a <= A4; a++) digitalWrite(a, a != quarter); digitalWrite(led, HIGH); 
}

void loop() {
    while (digitalRead(PushButton)); 
    unsigned long now = millis();
    while (!digitalRead(PushButton));
    unsigned long tijd = millis() - now;
    int Steps = 37 + random(18); 
    if (tijd > 500) Steps *= 2;  
    if (tijd > 5000) Steps *= 2.01;
    int Stap = 1;
    while (Stap < Steps) {
        Stap++;
        digitalWrite(led, LOW);
        if (CCW) {
            led++;
            if (led > 10) {
                if (quarter == A4) {
                    if (led > 11) {
                        quarter = A1;
                        led = 2;
                    }
                }
                else {
                    led = 2;
                    quarter++;
                }
            }
        }
        else { 
            led--;
            if (led < 2) {
                if (quarter == A1) {
                    led = ZeroLed;
                    quarter = A4;
                }
                else {
                    led = 10; 
                    quarter--;
                }
            }
        }
        for (int a = A1; a <= A4; a++) digitalWrite(a, a != quarter); 
        digitalWrite(led, HIGH);
        digitalWrite(Click, HIGH); delayMicroseconds(40); digitalWrite(Click, LOW);
        unsigned int delaytijd = 40 + Stap * 5;
        if (tijd > 500) delaytijd /= 2;
        delay(delaytijd);
    }
    CCW = !CCW;
}
